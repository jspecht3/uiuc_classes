\section{Heat Conduction}

\subsection{Fourier's Law}
\begin{equation}
    q'' = - k \nabla T
\end{equation}
\begin{equation}
    q'' = -k \frac{dT}{dx}
\end{equation}

When Fourier's law is used to compare the heat flux with Newton's law of cooling, the directions of the heat flux should go in the same direction. Newton's law of cooling goes from high temperature to low temperature.

\subsection{Thermal Conductivity}
It changes with temperature

\subsection{General Heat Diffusion Equation}
\begin{equation}
    \nabla \cdot k \nabla T + q''' = \rho c_p \frac{\partial T}{\partial t}
\end{equation}

This equation comes from saying the streaming of the heat in addition to the volumetric heat generation rate is equation to to the time rate of change of stored heat in a system.

For 1-D steady state.
\begin{equation}
    \frac{d^2T}{dx^2} = \frac{-q'''}{k}
\end{equation}

Common boundary conditions include finite temperature, equal heat fluxes at a boundary, and equal temperatures at a boundary.

\subsection{Treatment of Gap}
The gap can be handled as a region with Fourier's Law or as a resistance to heat transfer with the equation:

\begin{equation}
    R = \frac{1}{h \xi_h}
\end{equation}

\section{Convection: Fluids and Heat Transfer}
Mass:
\begin{equation}
    \frac{\partial \rho}{\partial t}  + \nabla \rho v = 0
\end{equation}

Momentum:
\begin{equation}
    \frac{\partial \rho v}{\partial t} + \nabla \rho v^2 = -\nabla P - \tau_f \frac{\xi_w}{A_f} - \rho g \sin(\theta)
\end{equation}

Energy:
\begin{equation}
    \frac{\partial (\rho h)}{\partial t} + \nabla \rho v h = \frac{q'' \xi_h}{A_f} + \frac{\partial P}{\partial t} + q'''
\end{equation}

\subsection{Approximations}
These equations are the time-averaged equations that result from assuming each variable is the sum of a mean and perturbed component.

Need additional closure relations.
\begin{equation}
    \tau_f = \frac{1}{2} f \rho v^2 = 
    \frac{1}{2} f \frac{G^2}{\rho}
\end{equation}

\subsection{Dimensionless Groups}

\begin{enumerate}
    \item Reynold's Number
    \begin{equation}
        Re = \frac{inertial\ forces}{viscous\ forces} = 
        \frac{\rho v L_c}{\mu} = 
        \frac{v L_c}{\nu} = 
        \frac{G L_c}{\mu}
    \end{equation}
    
    \item Prandtl Number
    \begin{equation}
        Pr = \frac{momentum\ diffusivity}{thermal\ diffusivity} =
        \frac{c_p \mu}{k} =
        \frac{\nu}{\alpha} =
        \frac{\delta}{\delta_t}
    \end{equation}

    \item Biot Number
    \begin{equation}
        Bi = \frac{\Delta T\ at\ boundary}{\Delta T\ in\ material} = 
        \frac{h L_c}{k_s}
    \end{equation}

    \item Fourier's Number
    \begin{equation}
        Fo = \frac{heat\ conducted}{heat\ stored} = 
        \frac{\alpha t}{L_c^2}
    \end{equation}

    \item Nusselt Number
    \begin{equation}
        Nu = \frac{convection\ at\ boundary}{conduction\ in\ fluid} = 
        \frac{h L_c}{k_f} =
        C Re^m Pr^n =
        C Gr^m Pr^n
    \end{equation}

    \item Peclet Number
    \begin{equation}
        Pe = \frac{advective\ transport\ rate}{mass\ diffusion\ rate} =
        RePr = 
        \frac{v L_c}{\alpha}
    \end{equation}

    \item Grashof Number
    \begin{equation}
        Gr = \frac{bouyant\ forces}{viscous\ forces} =
        \frac{g \beta \left(T_s - T_{\infty} \right) L_c^3}{\nu^2}
    \end{equation}

    \item Rayleigh Number
    \begin{equation}
        Ra = 
        \frac{advective\ transfer\ rate\ from\ bouyancy}{diffusive\ transfer\ rate} = 
        GrPr
    \end{equation}

    \item Jakob Number
    \begin{equation}
        Ja = \frac{sensible\ heat}{latent\ heat} = 
        \frac{c_p \Delta T}{h_{fg}} = 
        \frac{c_p \left(T_s - T_{sat} \right)}{h_{fg}}
    \end{equation}

    \item Bond/Eötvös Number
    \begin{equation}
        Bo = \frac{bouyant\ forces}{surface\ tension\ forces} =
        \frac{g \left(\rho_l - \rho_v \right) L_c^2}{\sigma}
    \end{equation}

\end{enumerate}
